    ```typescript
    import type { ImageProvider, ImageProviderInput, ImageProviderOutput } from "./types";
    import { IMAGE_GENERATION_MODEL, IMAGE_GENERATION_QUALITY, IMAGE_GENERATION_SIZE } from "@/lib/image-generation/config";

    export class OpenAIImageProvider implements ImageProvider {
      readonly name = "openai";
      private readonly model: string;

      constructor(model?: string) {
        this.model = model ?? IMAGE_GENERATION_MODEL;
      }

      async generateImage(input: ImageProviderInput): Promise<ImageProviderOutput> {
        const { default: OpenAI } = await import("openai");
        const openai = new OpenAI({
          apiKey: process.env.OPENAI_API_KEY,
        });

        const size = input.size ?? IMAGE_GENERATION_SIZE;
        const quality = input.quality ?? IMAGE_GENERATION_QUALITY;

        try {
          // Primary path: Responses API with image_generation tool
          const response = await openai.responses.create({
            model: this.model,
            input: [
              {
                role: "user",
                content: [
                  { type: "input_text", text: input.prompt },
                  ...(input.productImageDataUrl
                    ? [{ type: "input_image", image_url: input.productImageDataUrl }]
                    : []),
                ],
              },
            ],
            tools: [{
              type: "image_generation",
              size,
              quality,
            }],
          });

          // Extract the generated image from the response
          const imageOutput = response.output?.find(
            (item: any) => item.type === "image"
          );

          if (!imageOutput?.image_data?.url) {
            throw new Error("No image generated in Responses API response");
          }

          // Fetch the image and convert to base64
          const imageResponse = await fetch(imageOutput.image_data.url);
          const imageBuffer = await imageResponse.arrayBuffer();
          const imageBase64 = Buffer.from(imageBuffer).toString("base64");
          const contentType = imageResponse.headers.get("content-type") || "image/png";

          return {
            imageBase64,
            mimeType: contentType as "image/png" | "image/jpeg" | "image/webp",
            model: this.model,
          };
        } catch (err) {
          // Fallback: Image API edit when product image reference is needed
          // and Responses API path failed
          if (input.productImageDataUrl && this.isResponsesApiError(err)) {
            return this.fallbackToImageApi(openai, input, size);
          }
          throw err;
        }
      }

      private isResponsesApiError(err: unknown): boolean {
        const message = err instanceof Error ? err.message : String(err);
        // Treat as Responses API unavailability if message indicates
        // tool not supported, model not found, or specific feature error
        return (
          message.includes("not supported") ||
          message.includes("image_generation") ||
          message.includes("responses") ||
          message.includes("model_not_found") ||
          message.includes("tool")
        );
      }

      private async fallbackToImageApi(
        openai: any,
        input: ImageProviderInput,
        size: string
      ): Promise<ImageProviderOutput> {
        // Image API edit: requires a PNG with alpha channel as mask
        // We send the product image as the base image and the prompt as instruction
        const base64Data = input.productImageDataUrl!.replace(/^data:image\/\w+;base64,/, "");

        const response = await openai.images.edit({
          model: this.model,
          image: Buffer.from(base64Data, "base64"),
          prompt: input.prompt,
          size: size as "1024x1024" | "256x256" | "512x512",
          n: 1,
          response_format: "b64_json",
        });

        const imageBase64 = response.data[0]?.b64_json;
        if (!imageBase64) {
          throw new Error("Image API returned no image data");
        }

        return {
          imageBase64,
          mimeType: "image/png",
          model: this.model,
        };
      }
    }
    ```

    Key implementation notes:
    - Primary path: OpenAI Responses API with `image_generation` tool (preferred for conversational/editable image flows)
    - Fallback path: Image API edits when Responses API is unavailable for the specific use case
    - Product image passed as `input_image` content part in Responses API
    - Image result extracted from response output items
    - Error handling catches Responses API failures and falls back gracefully
    - The provider does NOT handle input validation — assumes validated input
    - Dynamic import of `openai` package to avoid ESM/CJS issues (same pattern as existing OpenAIProvider)

    ```typescript
    import { PromptLoader } from "@/lib/image-generation/prompt-loader";
    import { VISION_REVIEW_MODEL } from "@/lib/image-generation/config";
    import type { InputValidationResult } from "@/lib/image-generation/schema";

    export class InputValidationService {
      private readonly promptLoader: PromptLoader;
      private readonly model: string;

      constructor(promptLoader?: PromptLoader, model?: string) {
        this.promptLoader = promptLoader ?? new PromptLoader();
        this.model = model ?? VISION_REVIEW_MODEL;
      }

      async validate(
        typedProductName: string,
        productImageDataUrl: string,
        override?: { productImageCheck?: "user_confirmed_continue" }
      ): Promise<InputValidationResult> {
        // Skip validation when user override is present
        if (override?.productImageCheck === "user_confirmed_continue") {
          return { classification: "match", confidence: 1.0 };
        }

        // Load the prompt and interpolate the typed product name
        const prompt = this.promptLoader.load("campaign-input-visual-check", {
          typedProductName,
        });

        // Call the vision model
        const result = await this.callVisionModel(prompt, productImageDataUrl);

        // Parse and validate the structured JSON response
        return this.parseResult(result);
      }

      private async callVisionModel(
        prompt: string,
        imageDataUrl: string
      ): Promise<string> {
        const { default: OpenAI } = await import("openai");
        const openai = new OpenAI({
          apiKey: process.env.OPENAI_API_KEY,
        });

        const response = await openai.chat.completions.create({
          model: this.model,
          messages: [
            {
              role: "user",
              content: [
                { type: "text", text: prompt },
                {
                  type: "image_url",
                  image_url: { url: imageDataUrl, detail: "high" },
                },
              ],
            },
          ],
          max_tokens: 500,
        });

        const content = response.choices[0]?.message?.content;
        if (!content) {
          throw new Error("Vision model returned empty response");
        }

        return content;
      }

      private parseResult(raw: string): InputValidationResult {
        // Clean potential markdown code fences
        const cleaned = raw.replace(/```json\s*/g, "").replace(/```\s*/g, "").trim();
        const parsed = JSON.parse(cleaned);

        const classification = parsed.classification;

        switch (classification) {
          case "match":
            return {
              classification: "match",
              confidence: parsed.confidence ?? 1.0,
            };
          case "auto-fix":
            return {
              classification: "auto-fix",
              confidence: parsed.confidence ?? 0.9,
              correctedProductName: parsed.correctedProductName,
              reason: parsed.reason ?? "auto_fix",
            };
          case "conflict":
            return {
              classification: "conflict",
              confidence: parsed.confidence ?? 1.0,
              suggestedProductName: parsed.suggestedProductName,
              reason: parsed.reason ?? "Conflito entre nome digitado e imagem do produto",
            };
          case "low-confidence":
            return {
              classification: "low-confidence",
              confidence: parsed.confidence ?? 0.5,
              reason: parsed.reason ?? "Não foi possível confirmar a correspondência",
            };
          default:
            throw new Error(`Unknown classification: ${classification}`);
        }
      }
    }
    ```

    Key implementation notes:
    - Loads `campaign-input-visual-check.md` from PromptLoader with `typedProductName` variable
    - Sends the prompt + product image to a configured vision-capable text model (e.g., GPT-4o)
    - Parses the strict JSON response into the `InputValidationResult` union type
    - Skips validation entirely when `inputValidationOverride.productImageCheck === "user_confirmed_continue"`
    - Handles markdown code fence cleanup before JSON parsing
    - Export the class as named export

    ```typescript
    import { PromptLoader } from "@/lib/image-generation/prompt-loader";
    import { VISION_REVIEW_MODEL } from "@/lib/image-generation/config";
    import type { ImageReviewResult } from "@/lib/image-generation/schema";

    export interface ImageReviewInput {
      productName: string;
      storeName: string;
      originalPrice?: string;
      discountedPrice: string;
    }

    export class ImageReviewService {
      private readonly promptLoader: PromptLoader;
      private readonly model: string;

      constructor(promptLoader?: PromptLoader, model?: string) {
        this.promptLoader = promptLoader ?? new PromptLoader();
        this.model = model ?? VISION_REVIEW_MODEL;
      }

      async review(
        generatedImageDataUrl: string,
        input: ImageReviewInput
      ): Promise<ImageReviewResult> {
        // Load the review prompt with campaign data for comparison
        const prompt = this.promptLoader.load("campaign-image-reviewer", {
          productName: input.productName,
          storeName: input.storeName,
          discountedPrice: input.discountedPrice,
          originalPrice: input.originalPrice ?? "",
        });

        // Call the vision model with the generated image
        const raw = await this.callVisionModel(prompt, generatedImageDataUrl);

        // Parse the structured JSON response
        return this.parseResult(raw);
      }

      private async callVisionModel(
        prompt: string,
        imageDataUrl: string
      ): Promise<string> {
        const { default: OpenAI } = await import("openai");
        const openai = new OpenAI({
          apiKey: process.env.OPENAI_API_KEY,
        });

        const response = await openai.chat.completions.create({
          model: this.model,
          messages: [
            {
              role: "user",
              content: [
                { type: "text", text: prompt },
                {
                  type: "image_url",
                  image_url: { url: imageDataUrl, detail: "high" },
                },
              ],
            },
          ],
          max_tokens: 1000,
          response_format: { type: "json_object" },
        });

        const content = response.choices[0]?.message?.content;
        if (!content) {
          throw new Error("Vision model returned empty review");
        }

        return content;
      }

      private parseResult(raw: string): ImageReviewResult {
        const cleaned = raw.replace(/```json\s*/g, "").replace(/```\s*/g, "").trim();
        const parsed = JSON.parse(cleaned);

        const issues = (parsed.issues ?? []).map((issue: any) => ({
          type: issue.type,
          severity: issue.severity as "critical" | "minor",
          description: issue.description,
        }));

        return {
          passed: Boolean(parsed.passed),
          issues,
        };
      }
    }
    ```

    Key implementation notes:
    - Loads `campaign-image-reviewer.md` from PromptLoader with campaign data variables for comparison
    - Sends the generated image + review prompt to a configured vision-capable text model
    - Uses `response_format: { type: "json_object" }` to encourage structured output
    - Parses the response into `ImageReviewResult` with `passed` boolean and `issues[]` array
    - Handles markdown code fence cleanup before JSON parsing
    - Export the class as named export

### Task 04: Criar src/lib/auth/csrf.ts com requireSameOrigin

<action>
Criar `src/lib/auth/csrf.ts`:

```typescript
import { ForbiddenError } from "./errors";

export function requireSameOrigin(request: Request): void {
  const origin = request.headers.get("origin");
  const host = request.headers.get("host");
  const forwardedHost = request.headers.get("x-forwarded-host");

  if (!origin) {
    throw new ForbiddenError("Origin header required");
  }

  try {
    const originUrl = new URL(origin);
    const effectiveHost = forwardedHost || host;

    if (!effectiveHost || originUrl.host !== effectiveHost) {
      throw new ForbiddenError("Cross-origin request denied");
    }
  } catch {
    throw new ForbiddenError("Invalid origin");
  }
}
```

Regras:
- Origin ausente → ForbiddenError("Origin header required")
- Origin inválida (URL parse fail) → ForbiddenError("Invalid origin")
- Origin.host !== effectiveHost → ForbiddenError("Cross-origin request denied")
- x-forwarded-host tem precedência sobre host (proxy)
</action>

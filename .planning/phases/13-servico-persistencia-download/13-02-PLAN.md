```
import { NextRequest, NextResponse } from "next/server";
import { supabaseAdmin } from "@/lib/supabase/server";
import { requireApiUser } from "@/lib/auth/require-user";
import { requireOwnership } from "@/lib/auth/store-ownership";
import { apiHandler } from "@/lib/auth/api-handler";
import { notFound } from "@/lib/api-error-response";
import { getCampaign } from "@/lib/campaign/persistence";

const UUID_V4_REGEX = /^[0-9a-f]{8}-[0-9a-f]{4}-4[0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$/i;

export const GET = apiHandler(async (
  _request: NextRequest,
  { params }: { params: Promise<{ id: string }> }
) => {
  const { id } = await params;

  // Authenticate first (per OpenSpec pipeline: campaign-download-route/spec.md §L14)
  const user = await requireApiUser();

  // Validate UUID v4 after auth (per multitenant-campaign-guards/spec.md §L81)
  if (!UUID_V4_REGEX.test(id)) {
    return NextResponse.json({ error: "Invalid campaign ID" }, { status: 400 });
  }

  // Fetch campaign (supabaseAdmin — any tenant)
  const campaign = await getCampaign(id);
  if (!campaign) {
    return notFound("Campaign not found");
  }

  // Verify ownership (404 on mismatch — no existence leak)
  await requireOwnership(campaign.store_id, user.userId);

  // Generate signed URL (1 hour)
  const { data: signedData, error: signedError } = await supabaseAdmin
    .storage
    .from('campaign-images')
    .createSignedUrl(campaign.storage_path, 3600);

  if (signedError || !signedData?.signedUrl) {
    return NextResponse.json(
      { error: "Failed to generate download URL" },
      { status: 502 }
    );
  }

  // Redirect 302
  return NextResponse.redirect(signedData.signedUrl, 302);
});
```

Include all necessary imports, using the `getCampaign` helper imported from `@/lib/campaign/persistence`. The `apiHandler` wrapper will catch `UnauthorizedError` (→ 401) and `StoreNotFoundError` (→ 404). Auth (requireApiUser), UUID validation, campaign-not-found (404), ownership mismatch (404), and signed URL failure (502) are handled explicitly in the handler body.

**Order MUST be:** `requireApiUser()` → UUID validation → `getCampaign()` → `requireOwnership()` → `createSignedUrl()` → redirect 302. This preserves the OpenSpec contract: unauthenticated requests return 401 even with a malformed UUID.

Do NOT add `requireSameOrigin` — this is a GET redirect, not a mutation. The signed URL itself is the security mechanism.

  ```typescript
  import { notFound, redirect } from "next/navigation";
  import { requirePageUser } from "@/lib/auth/require-user";
  import { getCurrentStore } from "@/lib/auth/store-ownership";
  import { getCampaignForDisplay, generateSignedPreviewUrl, mapCampaignToProps } from "@/lib/campaign/display";
  import CampaignPageClient from "./client";

  interface PageProps {
    params: Promise<{ id: string }>;
  }

  export default async function CampaignPage({ params }: PageProps) {
    const { id } = await params;

    // 1. Authenticate
    const user = await requirePageUser();

    // 2. Resolve store ownership
    const store = await getCurrentStore(user.userId);
    if (!store) {
      redirect("/store");
    }

    // 3. Load campaign via RLS
    const campaign = await getCampaignForDisplay(id);
    if (!campaign) {
      notFound();
    }

    // 4. Compute display props
    const props = mapCampaignToProps(campaign, id);

    // 5. Generate signed URL only if ready
    if (campaign.status === "ready" && campaign.storage_path) {
      const signedUrl = await generateSignedPreviewUrl(campaign.storage_path);
      props.imageUrl = signedUrl;
    }

    return <CampaignPageClient {...props} />;
  }
  ```

  Key behaviors:
  - `requirePageUser()` redirects to `/login` if unauthenticated
  - `getCurrentStore(user.userId)` redirects to `/store` if no store exists
  - `getCampaignForDisplay(id)` calls `notFound()` if null (not owner or doesn't exist)
  - Signed URL generated only when `campaign.status === "ready"` AND `storage_path` is non-empty
  - All props are serializable — no functions, no Date objects passed to client
  </action>

  ```typescript
  "use client";

  import { useEffect, useCallback } from "react";
  import { useRouter } from "next/navigation";

  interface CampaignPageProps {
    imageUrl: string | null;
    caption: string;
    hashtags: string[];
    ctaPost: string;
    displayStatus: "ready" | "generating" | "stale" | "error";
    productName: string;
    createdAt: string;
    updatedAt: string;
    downloadUrl: string;
  }

  export default function CampaignPageClient(props: CampaignPageProps) {
    const router = useRouter();

    // Polling: refresh every 5s while generating
    useEffect(() => {
      if (props.displayStatus !== "generating") return;

      const interval = setInterval(() => {
        router.refresh();
      }, 5000);

      return () => clearInterval(interval);
    }, [props.displayStatus, router]);

    if (props.displayStatus === "generating") {
      return <GeneratingView />;
    }

    if (props.displayStatus === "stale") {
      return <StaleView onNewCampaign={() => router.push("/")} />;
    }

    if (props.displayStatus === "error") {
      return <ErrorView onNewCampaign={() => router.push("/")} />;
    }

    return <ReadyView {...props} />;
  }
  ```

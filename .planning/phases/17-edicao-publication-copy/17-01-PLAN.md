  **C. `validatePublicationCopy(body: unknown)` function:**

  Returns a discriminated union:
  - Success (edit): `{ valid: true; data: { caption: string; hashtags: string[]; cta_post: string } }`
  - Success (restore): `{ valid: true; data: { restore: true } }`
  - Failure: `{ valid: false; issues: ValidationIssue[] }`

  Validation rules:
  1. If `body` is null/undefined or not an object → `[{ field: "body", message: "Body inválido.", code: "invalid_body" }]`
  2. If `restore === true` → return valid immediately (skip all other validation)
  3. Must have `caption` (string, 1-2200 chars), `hashtags` (array, 0-30), `cta_post` (string, 0-200 chars)
  4. Collect ALL issues before returning (don't short-circuit on first error)
  5. Caption rules:
     - Must be a string → issue with code "invalid_type"
     - Must be 1-2200 chars → issue with code "too_short" if empty, "too_long" if > 2200
  6. Hashtags rules:
     - Must be an array → issue with code "invalid_type"
     - Max 30 items → issue with code "too_many"
     - Each item must start with `#` (regex: /^#\w+$/) → issue with code "invalid_format" (include index in field name like "hashtags[0]")
     - Each item must be 2-100 chars (including #) → issue with code "too_short" or "too_long"
  7. cta_post rules:
     - Must be a string → issue with code "invalid_type"
     - Max 200 chars → issue with code "too_long" (empty is valid)
  8. If no caption/hashtags/cta_post AND no restore → issue with code "invalid_body"

  Helper `isRecord(value: unknown): value is Record<string, unknown>`:
  ```typescript
  function isRecord(value: unknown): value is Record<string, unknown> {
    return typeof value === "object" && value !== null && !Array.isArray(value);
  }
  ```

  Messages in Portuguese:
  - `caption`: "Caption deve ter entre 1 e 2200 caracteres."
  - `hashtags`: "Hashtags deve ser um array de strings."
  - `hashtags`: "Máximo de 30 hashtags."
  - `hashtags[i]`: "Hashtag deve começar com # e conter apenas letras, números e underscore."
  - `cta_post`: "CTA deve ter no máximo 200 caracteres."
  - `body`: "Body deve conter caption, hashtags e cta_post para edição, ou restore: true para restaurar original."

  ```typescript
  export function getEffectivePublicationCopy(
    campaign: CampaignRecord
  ): { caption: string; hashtags: string[]; cta_post: string } {
    const current = campaign.publication_copy_current as Record<string, unknown> | null;
    const snapshot = campaign.publication_copy_snapshot as Record<string, unknown> | null;

    if (
      current &&
      typeof current.caption === "string" &&
      Array.isArray(current.hashtags) &&
      typeof current.cta_post === "string"
    ) {
      return {
        caption: current.caption,
        hashtags: current.hashtags as string[],
        cta_post: current.cta_post,
      };
    }

    if (
      snapshot &&
      typeof snapshot.caption === "string" &&
      Array.isArray(snapshot.hashtags) &&
      typeof snapshot.cta_post === "string"
    ) {
      return {
        caption: snapshot.caption,
        hashtags: snapshot.hashtags as string[],
        cta_post: snapshot.cta_post,
      };
    }

    return { caption: "", hashtags: [], cta_post: "" };
  }
  ```

```typescript
function getGreeting(storeName: string | null): string {
  const hour = new Date().getHours();
  const period =
    hour >= 6 && hour < 12 ? "Bom dia" :
    hour >= 12 && hour < 18 ? "Boa tarde" :
    "Boa noite";
  if (!storeName) return "Bem-vindo ao Vendeo";
  return `${period}, ${storeName}`;
}
```

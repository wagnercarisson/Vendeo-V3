<action>
Create `src/hooks/use-debounce.ts`:

```typescript
"use client";

import { useState, useEffect } from "react";

export function useDebounce<T>(value: T, delay: number): T {
  const [debouncedValue, setDebouncedValue] = useState<T>(value);

  useEffect(() => {
    const timer = setTimeout(() => {
      setDebouncedValue(value);
    }, delay);

    return () => {
      clearTimeout(timer);
    };
  }, [value, delay]);

  return debouncedValue;
}
```

- Uses `useState` + `useEffect` with `setTimeout`
- Cancels previous timeout on value change
- Cleans up on unmount
- Retorna valor sincronizado imediatamente na primeira renderização, depois apenas após `delay` ms sem mudanças
</action>

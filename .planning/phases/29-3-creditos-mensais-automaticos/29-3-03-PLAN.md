<action>
Create `src/app/api/cron/monthly-credits/route.ts`:

```typescript
import { NextRequest, NextResponse } from "next/server";
import { getLaunchConfig } from "@/lib/launch-config/config";
import { createServiceRoleClient } from "@/lib/supabase/admin";
import { logPipelineEvent } from "@/lib/pipeline-logger";

export async function GET(request: NextRequest) {
  // 1. Validate CRON_SECRET
  const authHeader = request.headers.get("authorization");
  const expectedToken = `Bearer ${process.env.CRON_SECRET}`;
  if (!authHeader || authHeader !== expectedToken) {
    return NextResponse.json({ error: "Unauthorized" }, { status: 401 });
  }

  // 2. Read Launch Config
  const config = getLaunchConfig();
  if (!config.monthlyCreditsEnabled) {
    return NextResponse.json({ skipped: true });
  }

  // 3. Execute RPC
  const supabase = createServiceRoleClient();
  const { data, error } = await supabase.rpc("grant_monthly_credits", {
    p_amount: config.monthlyCreditsAmount,
    p_bonus_cap: config.monthlyBonusCap,
    p_min_store_age_days: config.monthlyCreditsMinStoreAgeDays,
  });

  if (error) {
    await logPipelineEvent({
      event_type: "cron.monthly_credits",
      status: "error",
      metadata: { error: error.message },
    });
    return NextResponse.json({ error: error.message }, { status: 500 });
  }

  // 4. Log success
  await logPipelineEvent({
    event_type: "cron.monthly_credits",
    status: "success",
    metadata: data as Record<string, unknown>,
  });

  // 5. Return result
  return NextResponse.json(data);
}
```

Create `src/app/api/admin/monthly-credits/grant/route.ts`:

```typescript
import { NextRequest, NextResponse } from "next/server";
import { requireAdmin } from "@/lib/admin/gate";
import { getLaunchConfig } from "@/lib/launch-config/config";
import { createServiceRoleClient } from "@/lib/supabase/admin";
import { logPipelineEvent } from "@/lib/pipeline-logger";

export async function POST(request: NextRequest) {
  const admin = await requireAdmin();
  if (!admin) {
    return NextResponse.json({ error: "Unauthorized" }, { status: 403 });
  }

  const config = getLaunchConfig();
  if (!config.monthlyCreditsEnabled) {
    return NextResponse.json({ skipped: true });
  }

  const supabase = createServiceRoleClient();
  const { data, error } = await supabase.rpc("grant_monthly_credits", {
    p_amount: config.monthlyCreditsAmount,
    p_bonus_cap: config.monthlyBonusCap,
    p_min_store_age_days: config.monthlyCreditsMinStoreAgeDays,
  });

  if (error) {
    await logPipelineEvent({
      event_type: "admin.monthly_credits",
      status: "error",
      metadata: { actor_id: admin.id, error: error.message },
    });
    return NextResponse.json({ error: error.message }, { status: 500 });
  }

  await logPipelineEvent({
    event_type: "admin.monthly_credits",
    status: "success",
    metadata: { actor_id: admin.id, ...(data as Record<string, unknown>) },
  });

  return NextResponse.json(data);
}
```

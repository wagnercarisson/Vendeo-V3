   /**
    * Monta as variáveis contextuais do prompt de revisão.

    * Retorna Record com chaves: campaignIntentLabel, expectedPriceBehavior,
    * expectedBadgeBehavior, expectedImageTreatment, expectedCommercialTone,
    * productName, storeName, originalPrice, validationContextSection.

   buildReviewPromptVariables(input: ImageReviewInput): Record<string, string> {
     const intent = input.campaignIntent ?? "offer";
     // Etapa 1: resolver placeholders comerciais (já vêm formatados pelo pipeline)

    **Part B: Admin Exception Route**

    Criar `POST /api/admin/freemium/exception`:
    ```typescript
    import { NextRequest, NextResponse } from "next/server";
    import { supabaseAdmin } from "@/lib/supabase/server";
    import { requireAdmin } from "@/lib/admin/require-admin";
    import { apiHandler } from "@/lib/auth/api-handler";
    import { z } from "zod";

    const ExceptionSchema = z.object({
      storeId: z.string().uuid(),
      reason: z.string().min(10, "Motivo deve ter no mínimo 10 caracteres").max(500),
    });

    export const POST = apiHandler(async (request: NextRequest) => {
      await requireAdmin();
      const body = await request.json().catch(() => null);
      if (!body) return NextResponse.json({ error: "Invalid JSON body" }, { status: 400 });

      const parsed = ExceptionSchema.safeParse(body);
      if (!parsed.success) {
        return NextResponse.json(
          { error: parsed.error.errors.map(e => e.message).join("; ") },
          { status: 400 }
        );
      }

      const { storeId, reason } = parsed.data;

      // 1. Buscar store para obter root_hash
      const { data: store } = await supabaseAdmin
        .from("stores")
        .select("id, cnpj_root_hash")
        .eq("id", storeId)
        .single();

      if (!store) {
        return NextResponse.json({ error: "Loja não encontrada" }, { status: 404 });
      }

      const rootHash = store.cnpj_root_hash || "admin_exception_no_cnpj";

      // 2. Inserir entitlement admin_exception
      const { data: entitlement, error: entError } = await supabaseAdmin
        .from("freemium_entitlements")
        .insert({
          store_id: storeId,
          root_hash: rootHash,
          benefit_type: "admin_exception",
          reason: reason,
        })
        .select("id")
        .single();

      if (entError) {
        return NextResponse.json({ error: entError.message }, { status: 500 });
      }

      // 3. Conceder créditos (10 por padrão para admin exception)
      const { data: grantResult, error: grantError } = await supabaseAdmin.rpc("grant_credits", {
        p_store_id: storeId,
        p_amount: 10,
        p_reason: reason,
        p_idempotency_key: "admin_exception_" + entitlement.id,
        p_metadata: { source: "admin_exception", entitlement_id: entitlement.id },
        p_type: "admin_grant",
      });

      if (grantError) {
        return NextResponse.json({ error: grantError.message }, { status: 500 });
      }

      // 4. Atualizar grant_transaction_id no entitlement
      await supabaseAdmin
        .from("freemium_entitlements")
        .update({ grant_transaction_id: grantResult as string })
        .eq("id", entitlement.id);

      // 5. Registrar em admin_audit_log
      // Usa action='credit_grant' (valor existente na CHECK constraint) com grant_type='freemium_exception' no metadata
      await supabaseAdmin.from("admin_audit_log").insert({
        action: "credit_grant",
        target_type: "store",
        target_id: storeId,
        reason: reason,
        metadata: { grant_type: "freemium_exception", entitlement_id: entitlement.id, grant_transaction_id: grantResult },
      });

      return NextResponse.json({
        success: true,
        entitlementId: entitlement.id,
        grantTransactionId: grantResult,
      });
    });
    ```

```typescript
  value: number;

}
```
- `usd_brl_rate`: 1 USD = R$ X — converte custo estimado do provider (D1)

`EconomicParameterService` (server-only, padrão `OperationCostService` F38 — `src/lib/credit/operation-cost-service.ts`):
```typescript
import "server-only";
import type { SupabaseClient } from "@supabase/supabase-js";
import { supabaseAdmin } from "@/lib/supabase/server";
export class EconomicParameterUnavailableError extends Error { ... }  // fail-closed
export class EconomicParameterService {
  constructor(private readonly client: SupabaseClient = supabaseAdmin) {}
  async getParameter(key: EconomicParameterKey): Promise<EconomicParameterResolution>;
  async getAll(): Promise<EconomicParameterResolution[]>;
}
```
- `getParameter`: SELECT da linha em `economic_parameters` (via supabaseAdmin). Linha inexistente → `{ key, value: 1.00, source: "fallback" }` + `console.warn` (fail-open). **Erro real de leitura** (não é linha inexistente) → `throw new EconomicParameterUnavailableError(...)` (fail-closed → 503 na API).
- `getAll`: SELECT de todas as linhas + merge com fallback 1.00 para chaves ausentes (source visível para o admin).
- NUNCA altera a tabela — leitura apenas (escrita é via RPC `admin_set_economic_parameter`, 38-2-04).

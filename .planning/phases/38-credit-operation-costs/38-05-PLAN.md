`getAllCosts()` a adicionar em `src/lib/credit/operation-cost-service.ts` (ao lado de `getCost`, mesmo padrão D5):
```typescript
export interface AdminOperationCost {
  operationKey: OperationKey;
  costCredits: number;
  enabled: boolean;
  updatedBy: string | null;   // uuid do admin (NULL = seed de sistema)
  updatedAt: string | null;   // ISO (NULL = fallback)
  source: "table" | "fallback";
}
async getAllCosts(): Promise<AdminOperationCost[]> {
  const { data, error } = await this.client
    .from("credit_operation_costs")
    .select("operation_key, cost_credits, enabled, updated_by, updated_at")
    .in("operation_key", OPERATION_KEYS);   // cast se TS reclamar de readonly tuple
  if (error) { console.error("[operation-cost] getAllCosts error", error.message); throw new OperationCostUnavailableError(error.message); }
  // merge: chaves presentes → source "table" (updatedBy/updatedAt da linha);
  // chaves ausentes → DEFAULT_OPERATION_COSTS[key], updatedBy null, updatedAt null, source "fallback"
  // retorna na ordem do enum (OPERATION_KEYS), SEMPRE todas as chaves
}
```

import { PromptLoader } from '@/lib/image-generation/prompt-loader';
import { AiImageGenerator } from '@/lib/visual-signature/ai-image-generator';
import { uploadToStorage, persistSignature } from '@/lib/visual-signature/persistence';
import type {
  VisualSignatureWithoutLogoInput,
  VisualSignatureArtDirectorOutput,
  VisualSignatureRecord,
} from '@/lib/visual-signature/types';

export interface VisualSignatureGenerationResult {
  signature: VisualSignatureRecord;
  artDirectorOutput: VisualSignatureArtDirectorOutput;
  assetUrl: string;
  referenceCardUrl?: string;
}

export class StoreIdentityArtDirectorService {
  private promptLoader: PromptLoader;
  private aiImageGenerator: AiImageGenerator;

  constructor() {
    this.promptLoader = new PromptLoader();
    this.aiImageGenerator = new AiImageGenerator();
  }

  async generate(input: VisualSignatureWithoutLogoInput, signal?: AbortSignal): Promise<VisualSignatureGenerationResult> {
    // 1. Load prompt with store data (using input fields directly as variables)
    const prompt = this.promptLoader.load('store-identity-art-director', {
      storeName: input.storeName,
      segment: input.segment,
      subsegment: input.subsegment ?? '',
      tone_of_voice: input.tone_of_voice ?? '',
      positioning: input.positioning ?? '',
      short_description: input.short_description ?? '',
      slogan: input.slogan ?? '',
      city: input.city ?? '',
      state: input.state ?? '',
      brandColor: input.brandColor ?? '#22C55E',
      rejectionContext: input.rejectionContext
        ? `A versão anterior (tentativa ${input.rejectionContext.attempt}) foi rejeitada porque: ${input.rejectionContext.reason}. Busque uma direção criativa completamente diferente — não apenas uma variação pequena da anterior.`
        : '',
    });

    // 2. Try image generation via AiImageGenerator (attempt 1)
    const tone = input.tone_of_voice ?? 'profissional';
    try {
      const result = await this.aiImageGenerator.generate({
        storeId: input.storeId,
        storeName: input.storeName,
        segment: input.segment,
        brandColor: input.brandColor ?? '#22C55E',
        tone,
        signal,
        attempt: 1,
      });

      // TODO: Extract creative metadata from the generation
      // For V1, construct metadata from the prompt context
      const artDirectorOutput: VisualSignatureArtDirectorOutput = {
        creative_description: `Assinatura visual para ${input.storeName} (${input.segment})`,
        suggested_colors: input.brandColor ? [input.brandColor] : ['#22C55E'],
        visual_direction: 'Personalizada',
        elements_used: ['nome da loja'],
      };

      // 3. TODO: Generate reference card (separate image generation or composite)
      // For V1, the visual signature itself serves as reference

      return {
        signature: null as unknown as VisualSignatureRecord,
        artDirectorOutput,
        assetUrl: result.assetUrl,
      };
    } catch (error) {
      // 4. Retry with simplified prompt (attempt 2) handled by existing cascade
      throw error;
    }
  }
}
```

**Implementation notes:**
- The `StoreIdentityArtDirectorService` orchestrates the generation of visual signatures for stores without logo
- It reuses `AiImageGenerator` from the existing visual-signature subsystem (which already has its own cascade: image_direct → image_retry → typographic fallback)
- For V1, the identity art director delegates to `AiImageGenerator` with the store data as creative direction
- Creative metadata extraction: since the Responses API returns the image directly, the creative description, suggested colors, and visual direction are constructed from the input data + prompt context
- The reference card is deferred to a future enhancement; the visual signature URL is returned as the primary asset
- Error handling follows the existing cascade pattern: on timeout/error, `AiImageGenerator` retries once with simplified prompt

**Actual implementation approach:**
1. Create the class with constructor that initializes PromptLoader and AiImageGenerator
2. `generate()` method: builds prompt with rejection context if present, calls AiImageGenerator.generate(), constructs VisualSignatureArtDirectorOutput from input data, persists the signature record via persistSignature(), returns VisualSignatureGenerationResult
3. Error: wrap AI errors in a controlled error message (no typographic fallback per D-12)

import { supabaseAdmin as supabase } from '@/lib/supabase/server';
import type { GenerationEventInsert, GenerationEventRecord } from '@/lib/visual-signature/types';

export async function insertGenerationEvent(
  event: GenerationEventInsert
): Promise<GenerationEventRecord | null> {
  try {
    const { data, error } = await supabase
      .from('generation_events')
      .insert(event)
      .select()
      .single();

    if (error) {
      console.error('[GenerationEvents] Insert failed (best-effort):', error.message);
      return null;
    }

    return data;
  } catch (err) {
    console.error('[GenerationEvents] Insert exception (best-effort):', err);
    return null;
  }
}

export async function updateGenerationEventDecision(
  assetId: string,
  attemptNumber: number,
  decision: { approved?: boolean; rejected?: boolean }
): Promise<void> {
  try {
    await supabase
      .from('generation_events')
      .update(decision)
      .eq('asset_id', assetId)
      .eq('attempt_number', attemptNumber);
  } catch (err) {
    console.error('[GenerationEvents] Update failed (best-effort):', err);
  }
}
```

Both functions are best-effort — they catch all errors and log them without throwing. The generation pipeline MUST continue even if metrics recording fails.

2. Create shared validation helper (or use inline) for subsegment:
```typescript
const GENERIC_SUBSEGMENT_VALUES = ["outro", "loja", "comercio", "comércio", "varejo"];

function validateSubsegment(value: string): string | null {
  const trimmed = value.trim();
  if (trimmed.length < 3) return "Digite ao menos 3 caracteres";
  if (trimmed.length > 30) return "Máximo de 30 caracteres";
  if (!/^[A-Za-zÀ-ü\s]+$/.test(trimmed)) return "Use apenas letras e espaços";
  if (GENERIC_SUBSEGMENT_VALUES.includes(trimmed.toLowerCase())) return "Valor genérico não permitido";
  return null;
}

function sanitizeSubsegment(value: string): string {
  return value
    .trim()
    .replace(/\s+/g, " ")
    .split(" ")
    .map(w => w.charAt(0).toUpperCase() + w.slice(1).toLowerCase())
    .join(" ");
}
```

3. Update segment validation: replace check against `VALID_SEGMENTS` with:
```typescript
const validSegments = STORE_SEGMENTS.map(s => s.value);
if (!validSegments.includes(body.segment)) {
  return NextResponse.json({ error: "Segmento inválido" }, { status: 400 });
}
```

4. Add subsegment validation logic:
- If `segment === "outros"` and `subsegment` is empty/missing → return 400 "Subsegmento obrigatório para segmento outros"
- If subsegment is provided and is literal `"outro"` → return 400 "Valor inválido para subsegmento"
- If subsegment is provided → run validateSubsegment(); if error → return 400 with error message
- If subsegment passes validation → apply sanitizeSubsegment() before persisting

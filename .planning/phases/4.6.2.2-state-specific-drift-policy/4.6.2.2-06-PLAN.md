```typescript
const handleStep2Submit = async () => {
  // ... existing validation ...

  const { driftCategory, criticalDrift, dismissCriticalDrift } = getDriftDetection()

  if (driftCategory === 'critical' && criticalDrift?.status === 'new') {
    setShowDriftCriticalModal(true)
    return  // não salva ainda, espera decisão do modal
  }

  if (driftCategory === 'sensitive') {
    setShowDriftDecisionModal(true)
    return  // não salva ainda
  }

  // Sem drift: salvar direto
  await executeStep2Save()
}
```

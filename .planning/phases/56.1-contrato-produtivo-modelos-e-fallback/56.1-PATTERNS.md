**Cache with TTL + in-flight dedupe + explicit invalidation** (lines 28-88):
```typescript
export class AiModelSelectionService {
  private cache: { expiresAt: number; selections: AiModelSelectionMap } | null = null;
  private inFlight: { epoch: number; promise: Promise<AiModelSelectionMap> } | null = null;
  private invalidationEpoch = 0;

  constructor(
    private readonly client: SupabaseClient = supabaseAdmin,
    private readonly ttlMs = 30_000,
    private readonly now: () => number = () => Date.now(),
  ) {}

  async getSelectionMap(): Promise<AiModelSelectionMap> {
    if (this.cache && this.cache.expiresAt > this.now()) return this.cache.selections;
    const epoch = this.invalidationEpoch;
    if (this.inFlight?.epoch === epoch) return this.inFlight.promise;
    const promise = this.loadSelectionMap();
    this.inFlight = { epoch, promise };
    try {
      const selections = await promise;
      if (epoch === this.invalidationEpoch && this.inFlight?.promise === promise) {
        this.cache = { selections, expiresAt: this.now() + this.ttlMs };
      }
      return selections;
    } finally {
      if (this.inFlight?.promise === promise) this.inFlight = null;
    }
  }

  invalidateModelSelectionCache(): void {
    this.invalidationEpoch += 1;
    this.cache = null;
    this.inFlight = null;
  }
}
```
Export a singleton + module wrappers + `invalidateXxxCache()` exactly like lines 86-98. **Difference from the analog:** the legacy service logs and returns an empty map (fail-open, line 73/80). The new-flow resolver MUST **fail closed** (D-06): missing/invalid/divergent config throws an identifiable error, never a silent default.

function invalidBody(error: ZodError): Response {

function rpcError(error: { message: string }): Response {

    { status: badRequestCodes.some((code) => error.message.includes(code)) ? 400 : 500 });

export const PUT = apiHandler(async (request: Request) => {

  let body;

  catch (error) {
    if (error instanceof ZodError) return invalidBody(error);
    return NextResponse.json({ error: "Corpo da requisição inválido" }, { status: 400 });

    p_actor_id: admin.userId, p_operation_id: body.operationId,

});

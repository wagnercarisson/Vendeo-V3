```typescript
export function resolutionErrorStatus(error: unknown): number {
  const code = error instanceof PendingOperationResolutionError
    ? error.code
    : ((error as { code?: string } | null)?.code ?? "unknown");
  if (BAD_REQUEST_CODES.includes(code)) return 400;      // missing_reason, missing_actor_id,
                                                        // missing_campaign_id, missing_operation_id,
                                                        // missing_resolution_operation_id,
                                                        // invalid_action, invalid_evidence,
                                                        // insufficient_evidence
  if (code === "operation_not_found") return 404;
  if (CONFLICT_CODES.includes(code)) return 409;         // operation_terminal, operation_in_progress,
                                                        // execution_not_ended, delivered_not_refundable,
                                                        // operation_identity_conflict
  return 500;                                            // técnico/desconhecido
}
```
